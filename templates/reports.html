db.transactions.aggregate([
  // 1. Filter by owner user, expense type, and month date range
  { $match: {
      user_id: "{{ user.id }}",
      type: "expense",
      transaction_date: {
        $gte: ISODate("{{ selected_year }}-{{ selected_month|stringformat:'02d' }}-01T00:00:00Z"),
        $lt: ISODate(...)
      }
  }},
  // 2. Group by category and compute sum of Decimal128 amount
  { $group: {
      _id: "$category_id",
      totalSpent: { $sum: "$amount" },
      count: { $sum: 1 }
  }},
  // 3. Join with categories collection for referenced category name
  { $lookup: {
      from: "categories",
      localField: "_id",
      foreignField: "_id",
      as: "category_info"
  }},
  // 4. Sort descending by highest expense
  { $sort: { totalSpent: -1 } }
